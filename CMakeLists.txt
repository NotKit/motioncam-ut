cmake_minimum_required(VERSION 3.16)
project(motioncam CXX C)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)

find_package(PkgConfig REQUIRED)
find_package(Qt5 REQUIRED COMPONENTS Core Gui Quick Qml Concurrent)
# opencv4.pc is only in the meta-package libopencv-dev (not individual component
# packages); fall back to explicit paths when it's absent.
pkg_check_modules(OpenCV opencv4)
if(NOT OpenCV_FOUND)
    set(OpenCV_INCLUDE_DIRS /usr/include/opencv4)
    set(OpenCV_LIBRARIES opencv_core opencv_imgproc opencv_photo opencv_video)
    set(OpenCV_FOUND TRUE)
endif()

pkg_check_modules(EGL    REQUIRED egl)
pkg_check_modules(GLES   REQUIRED glesv2)
pkg_check_modules(ZSTD   REQUIRED libzstd)

find_library(HYBRIS_COMMON_LIB hybris-common REQUIRED)
set(HYBRIS_INCLUDE_DIRS /usr/include)

execute_process(
    COMMAND dpkg-architecture -qDEB_HOST_MULTIARCH
    OUTPUT_VARIABLE ARCH_TRIPLET
    OUTPUT_STRIP_TRAILING_WHITESPACE
)

set(DATA_DIR /)
configure_file(manifest.json.in ${CMAKE_CURRENT_BINARY_DIR}/manifest.json)

# ── Paths ──────────────────────────────────────────────────────────────────────

set(LIBMOTIONCAM_ROOT   ${CMAKE_SOURCE_DIR}/libMotionCam)
set(LIBMOTIONCAM_SRC    ${LIBMOTIONCAM_ROOT}/libMotionCam)
set(THIRDPARTY          ${LIBMOTIONCAM_ROOT}/thirdparty)

# ── DNG SDK (from source) ──────────────────────────────────────────────────────

file(GLOB DNG_SDK_SOURCES ${THIRDPARTY}/dng_sdk/dng/*.cpp)
# dng_xmp_sdk.cpp unconditionally includes XMP.incl_cpp (Adobe XMP SDK, not vendored)
list(REMOVE_ITEM DNG_SDK_SOURCES ${THIRDPARTY}/dng_sdk/dng/dng_xmp_sdk.cpp)

add_library(dng-sdk STATIC ${DNG_SDK_SOURCES})

target_include_directories(dng-sdk PUBLIC ${THIRDPARTY}/dng_sdk)

target_compile_definitions(dng-sdk PUBLIC
    qDNGThreadSafe=1
    qDNGUseLibJPEG=0
    qDNGUseXMP=0
    qDNGValidate=0
)

target_compile_options(dng-sdk PRIVATE -Wno-reorder -Wno-unused-parameter -Wno-sign-compare)

# ── Halide kernels (generated by scripts/generate_halide_kernels.sh) ──────────
set(HALIDE_KERNELS_DIR "${LIBMOTIONCAM_SRC}/halide/arm64-linux")
set(HALIDE_INCLUDE_DIR "${LIBMOTIONCAM_SRC}/halide/include")

if(NOT EXISTS "${HALIDE_KERNELS_DIR}/.done")
    message(STATUS "Halide kernels not present — running generate_halide_kernels.sh...")
    execute_process(
        COMMAND bash "${CMAKE_SOURCE_DIR}/scripts/generate_halide_kernels.sh"
        WORKING_DIRECTORY "${CMAKE_SOURCE_DIR}"
        RESULT_VARIABLE _halide_gen_result
        OUTPUT_VARIABLE _halide_gen_out
        ERROR_VARIABLE  _halide_gen_err
    )
    if(NOT _halide_gen_result EQUAL 0)
        message(WARNING
            "Halide kernel generation failed (exit ${_halide_gen_result}) — "
            "ImageProcessor will be disabled.\n${_halide_gen_err}")
    endif()
endif()

if(EXISTS "${HALIDE_KERNELS_DIR}/.done")
    message(STATUS "Halide kernels found — ImageProcessor enabled")
    set(HAVE_HALIDE TRUE)

    pkg_check_modules(Exiv2 exiv2)
    if(NOT Exiv2_FOUND)
        find_library(Exiv2_LIBRARIES exiv2 REQUIRED)
        set(Exiv2_INCLUDE_DIRS /usr/include)
    endif()

    set(HALIDE_KERNEL_NAMES
        fuse_denoise_3x3 fuse_denoise_5x5 fuse_denoise_7x7
        forward_transform fuse_image inverse_transform
        generate_stats measure_noise build_bayer build_bayer2
        hdr_mask linear_image measure_image generate_edges
        deinterleave_raw postprocess fast_preview fast_preview2
        preview_landscape2 preview_portrait2
        preview_reverse_portrait2 preview_reverse_landscape2
        preview_landscape4 preview_portrait4
        preview_reverse_portrait4 preview_reverse_landscape4
        preview_landscape8 preview_portrait8
        preview_reverse_portrait8 preview_reverse_landscape8
        halide_runtime_host
    )
    set(HALIDE_IMPORTED_LIBS "")
    foreach(K ${HALIDE_KERNEL_NAMES})
        add_library(halide::${K} STATIC IMPORTED)
        set_target_properties(halide::${K} PROPERTIES
            IMPORTED_LOCATION "${HALIDE_KERNELS_DIR}/${K}.a")
        list(APPEND HALIDE_IMPORTED_LIBS halide::${K})
    endforeach()
else()
    message(STATUS "Halide kernels not found — ImageProcessor disabled (run scripts/generate_halide_kernels.sh)")
    set(HAVE_HALIDE FALSE)
endif()

# ── libMotionCam ───────────────────────────────────────────────────────────────

add_library(motion-cam STATIC
    ${THIRDPARTY}/json11/json11/json11.cpp
    ${THIRDPARTY}/miniz/miniz.c
    ${THIRDPARTY}/miniz/miniz_zip.c
    ${THIRDPARTY}/miniz/miniz_tinfl.c
    ${THIRDPARTY}/miniz/miniz_tdef.c
    ${THIRDPARTY}/tinywav/tinywav.c
    ${LIBMOTIONCAM_SRC}/source/CameraProfile.cpp
    ${LIBMOTIONCAM_SRC}/source/Color.cpp
    ${LIBMOTIONCAM_SRC}/source/Logger.cpp
    ${LIBMOTIONCAM_SRC}/source/Measure.cpp
    ${LIBMOTIONCAM_SRC}/source/RawBufferManager.cpp
    ${LIBMOTIONCAM_SRC}/source/RawBufferStreamer.cpp
    ${LIBMOTIONCAM_SRC}/source/RawCameraMetadata.cpp
    ${LIBMOTIONCAM_SRC}/source/RawContainer.cpp
    ${LIBMOTIONCAM_SRC}/source/RawContainerImpl.cpp
    ${LIBMOTIONCAM_SRC}/source/RawContainerImpl_Legacy.cpp
    ${LIBMOTIONCAM_SRC}/source/RawEncoder.cpp
    ${LIBMOTIONCAM_SRC}/source/RawImageBuffer.cpp
    ${LIBMOTIONCAM_SRC}/source/Settings.cpp
    ${LIBMOTIONCAM_SRC}/source/Temperature.cpp
    ${LIBMOTIONCAM_SRC}/source/Util.cpp
)

if(HAVE_HALIDE)
    target_sources(motion-cam PRIVATE
        ${LIBMOTIONCAM_SRC}/source/ImageOps.cpp
        ${LIBMOTIONCAM_SRC}/source/ImageProcessor.cpp
        ${LIBMOTIONCAM_SRC}/source/MotionCam.cpp
        ${LIBMOTIONCAM_SRC}/source/imgcodecs_impl.cpp
    )
endif()

target_include_directories(motion-cam PUBLIC
    # android-headers: shims opencv2/opencv.hpp and (when no Halide) HalideBuffer.h
    ${CMAKE_SOURCE_DIR}/android-headers
    ${LIBMOTIONCAM_SRC}/include
    ${THIRDPARTY}/json11
    ${THIRDPARTY}/miniz
    ${THIRDPARTY}/tinywav
    ${THIRDPARTY}/queue
    ${THIRDPARTY}/atomic_queue
    ${THIRDPARTY}/dng_sdk
    ${THIRDPARTY}/zstd/include
    ${OpenCV_INCLUDE_DIRS}
)

if(HAVE_HALIDE)
    # Real HalideBuffer.h must precede the stub in android-headers
    target_include_directories(motion-cam BEFORE PUBLIC
        ${HALIDE_INCLUDE_DIR}
        ${HALIDE_KERNELS_DIR}
        ${Exiv2_INCLUDE_DIRS}
    )
    target_compile_definitions(motion-cam PUBLIC HAVE_IMAGE_PROCESSOR)
endif()

target_compile_definitions(motion-cam PUBLIC
    DNG_SUPPORT
    qDNGThreadSafe=1
    qDNGUseLibJPEG=0
    qDNGUseXMP=0
    qDNGValidate=0
)

target_compile_options(motion-cam PRIVATE
    -Wno-unused-parameter -Wno-sign-compare -Wno-reorder
)

# RawEncoder.cpp is compiled from source in the motion-cam target above.
# The prebuilt libmotioncam-encoder.a (Android NDK Bionic/libc++) is NOT used.

target_link_libraries(motion-cam
    dng-sdk
    ${OpenCV_LIBRARIES}
    ${ZSTD_LIBRARIES}
)

if(HAVE_HALIDE)
    target_link_libraries(motion-cam
        ${Exiv2_LIBRARIES}
        ${HALIDE_IMPORTED_LIBS}
        dl      # Halide runtime uses dlsym; must follow the archives on the link line
        jpeg    # imgcodecs_impl.cpp
        png16   # imgcodecs_impl.cpp
    )
endif()

# ── Qt resources ───────────────────────────────────────────────────────────────

qt5_add_resources(QT_RESOURCES qml/qml.qrc)

# ── Main executable ────────────────────────────────────────────────────────────

add_executable(motioncam
    # NDK loader + Camera2 NDK shim
    src/ndk_loader.cpp
    src/camera2ndk_shim.cpp

    # Camera session layer (adapted from motioncam Android, vtable via shim)
    src/camera/CameraSession.cpp
    src/camera/CameraStateManager.cpp
    src/camera/CaptureSessionManager.cpp
    src/camera/RawImageConsumer.cpp
    src/camera/ClHelper.cpp
    src/camera/NativeClBuffer.cpp

    # Qt bridge + UI
    src/burst_preview_provider.cpp
    src/camera_bridge.cpp
    src/main.cpp

    ${QT_RESOURCES}
)

if(HAVE_HALIDE)
    # Real HalideBuffer.h must precede the stub in android-headers for the
    # motioncam executable TUs that include ImageProcessor.h transitively.
    target_include_directories(motioncam BEFORE PRIVATE
        ${HALIDE_INCLUDE_DIR}
        ${HALIDE_KERNELS_DIR}
    )
endif()

target_include_directories(motioncam PRIVATE
    src/
    ${CMAKE_SOURCE_DIR}/android-headers
    ${HYBRIS_INCLUDE_DIRS}
    ${EGL_INCLUDE_DIRS}
    ${GLES_INCLUDE_DIRS}
    ${LIBMOTIONCAM_SRC}/include
    ${THIRDPARTY}/json11
    ${THIRDPARTY}/miniz
    ${THIRDPARTY}/queue
    ${THIRDPARTY}/atomic_queue
    ${THIRDPARTY}/dng_sdk
    ${THIRDPARTY}/zstd/include
    ${OpenCV_INCLUDE_DIRS}
)

target_compile_definitions(motioncam PRIVATE
    DNG_SUPPORT
    qDNGThreadSafe=1
    qDNGUseLibJPEG=0
    qDNGUseXMP=0
    qDNGValidate=0
)

# Inject android_compat.h before all TUs so AOSP NDK availability macros
# and nullability annotations are defined on the Linux toolchain.
target_compile_options(motioncam PRIVATE
    -Wall -Wextra -Wno-unused-parameter -Wno-sign-compare
    -include ${CMAKE_SOURCE_DIR}/android-headers/android_compat.h
    -DWITH_QT
)

target_link_libraries(motioncam
    Qt5::Core Qt5::Gui Qt5::Quick Qt5::Qml Qt5::Concurrent
    ${HYBRIS_COMMON_LIB}
    ${EGL_LIBRARIES}
    ${GLES_LIBRARIES}
    ${OpenCV_LIBRARIES}
    ${ZSTD_LIBRARIES}
    motion-cam
    dng-sdk
    pthread
    z
)

set_target_properties(motioncam PROPERTIES LINK_FLAGS_RELEASE -s)

# ── Install ────────────────────────────────────────────────────────────────────

install(TARGETS motioncam RUNTIME DESTINATION ${CMAKE_INSTALL_PREFIX})
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/manifest.json DESTINATION ${CMAKE_INSTALL_PREFIX})
install(FILES motioncam.apparmor DESTINATION ${DATA_DIR})
install(FILES motioncam.desktop  DESTINATION ${DATA_DIR})
install(FILES assets/motioncam.png DESTINATION ${DATA_DIR})
